name: Issue guard

on:
  pull_request:
    paths:
      - tools/issue_guard.py
      - tests/governance/**
      - packages/ci-controls/**
      - .github/workflows/issue-guard.yml
      - .github/ISSUE_TEMPLATE/**
      - docs/operating-model.md
      - pyproject.toml
      - uv.lock
  push:
    branches: [main]
  issues:
    types: [opened, edited, closed, reopened, deleted, transferred]
  schedule:
    - cron: "17 * * * *"
  workflow_dispatch:

permissions: {}

concurrency:
  group: issue-guard-${{ github.repository }}
  cancel-in-progress: false

jobs:
  validate:
    runs-on: ubuntu-24.04
    timeout-minutes: 10
    permissions:
      contents: read
    env:
      UV_LINK_MODE: copy
      UV_NO_PROGRESS: "1"
    steps:
      - name: Check out the exact candidate
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
      - name: Install Python
        uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97 # v7.0.0
        with:
          python-version: "3.13"
      - name: Install uv
        uses: astral-sh/setup-uv@c771a70e6277c0a99b617c7a806ffedaca235ff9 # v9.0.0
        with:
          version: "0.11.32"
          enable-cache: false
      - name: Validate with fake clients and no Issues token
        run: |
          uv lock --check
          uv sync --frozen --all-groups --all-packages
          uv run --frozen --no-sync pytest -q --disable-socket tests/governance

  reconcile:
    if: github.event_name != 'pull_request'
    needs: validate
    runs-on: ubuntu-24.04
    timeout-minutes: 10
    permissions:
      contents: read
      issues: write
    env:
      UV_LINK_MODE: copy
      UV_NO_PROGRESS: "1"
    steps:
      - name: Check out the default-branch implementation
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
      - name: Install Python
        uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97 # v7.0.0
        with:
          python-version: "3.13"
      - name: Install uv
        uses: astral-sh/setup-uv@c771a70e6277c0a99b617c7a806ffedaca235ff9 # v9.0.0
        with:
          version: "0.11.32"
          enable-cache: false
      - name: Reconcile body authority to native dependencies
        env:
          GITHUB_TOKEN: ${{ github.token }}
        run: |
          uv lock --check
          uv sync --frozen --all-groups --all-packages
          uv run --frozen --no-sync python tools/issue_guard.py reconcile --repository "${{ github.repository }}" --format json --max-pages 20 --timeout-seconds 20
