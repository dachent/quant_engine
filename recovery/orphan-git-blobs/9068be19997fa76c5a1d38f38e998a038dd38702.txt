"""Fail-closed HAR-first routing and API credit evidence for Portfolio123."""

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from enum import StrEnum
from types import MappingProxyType
from typing import Mapping


API_DOCUMENTATION_URL = "https://api.portfolio123.com/"
API_COST_DOCUMENTATION_URL = (
    "https://portfolio123.customerly.help/en/articles/13766-api-credits"
)


class Portfolio123RoutingError(ValueError):
    """A capability request violates the reviewed routing or approval contract."""


class Transport(StrEnum):
    HAR = "HAR"
    API = "API"


class RouteReason(StrEnum):
    HAR_FIRST = "HAR_FIRST"
    UNSUPPORTED_BY_HAR = "UNSUPPORTED_BY_HAR"
    USER_REQUESTED_API = "USER_REQUESTED_API"


class ApiEffect(StrEnum):
    METERED_READ = "METERED_READ"
    PROVIDER_MUTATION = "PROVIDER_MUTATION"


@dataclass(frozen=True)
class CapabilityPolicy:
    """Reviewed transports and external effect for one user-facing capability."""

    name: str
    har_supported: bool
    api_operation: str | None
    api_effect: ApiEffect | None


@dataclass(frozen=True)
class ApiAuthorization:
    """Issue-recorded authorization for exactly one metered API run."""

    operation: str
    estimated_credits: int
    authorized_credit_budget: int
    approval_reference: str
    credit_table_captured_at: datetime
    documentation_url: str = API_DOCUMENTATION_URL
    cost_documentation_url: str = API_COST_DOCUMENTATION_URL

    def validate(self) -> None:
        if not self.operation.strip():
            raise Portfolio123RoutingError("API operation is required")
        if self.estimated_credits <= 0 or self.authorized_credit_budget <= 0:
            raise Portfolio123RoutingError("API credit values must be positive")
        if self.estimated_credits > self.authorized_credit_budget:
            raise Portfolio123RoutingError("estimated API cost exceeds the budget")
        if not self.approval_reference.strip():
            raise Portfolio123RoutingError("API approval reference is required")
        if self.credit_table_captured_at.tzinfo is None:
            raise Portfolio123RoutingError("credit-table time must be timezone-aware")
        if self.documentation_url != API_DOCUMENTATION_URL:
            raise Portfolio123RoutingError("API documentation anchor is not canonical")
        if self.cost_documentation_url != API_COST_DOCUMENTATION_URL:
            raise Portfolio123RoutingError("API cost anchor is not canonical")


@dataclass(frozen=True)
class RouteDecision:
    capability: str
    transport: Transport
    reason: RouteReason
    api_operation: str | None = None
    authorization: ApiAuthorization | None = None
    provider_mutation_approval: str | None = None


@dataclass(frozen=True)
class ApiRunEvidence:
    """Post-run cost evidence retained beside the source run."""

    decision: RouteDecision
    actual_credits: int
    quota_remaining: int

def capability_policy(capability: str) -> CapabilityPolicy:
    try:
        return _CAPABILITIES[capability]
    except KeyError as exc:
        raise Portfolio123RoutingError("capability is not reviewed") from exc


def route_capability(
    capability: str,
    *,
    user_requested_api: bool = False,
    api_authorization: ApiAuthorization | None = None,
    provider_mutation_approval: str | None = None,
) -> RouteDecision:
    """Choose HAR first; API requires declared need, cost evidence, and approval."""

    policy = capability_policy(capability)
    if policy.har_supported and not user_requested_api:
        return RouteDecision(capability, Transport.HAR, RouteReason.HAR_FIRST)
    if policy.api_operation is None or policy.api_effect is None:
        raise Portfolio123RoutingError("capability has no reviewed API route")
    reason = (
        RouteReason.USER_REQUESTED_API
        if user_requested_api
        else RouteReason.UNSUPPORTED_BY_HAR
    )
    if api_authorization is None:
        raise Portfolio123RoutingError("metered API authorization is required")
    api_authorization.validate()
    if api_authorization.operation != policy.api_operation:
        raise Portfolio123RoutingError("authorization operation does not match capability")
    if policy.api_effect is ApiEffect.PROVIDER_MUTATION:
        if not provider_mutation_approval or not provider_mutation_approval.strip():
            raise Portfolio123RoutingError(
                "provider mutation requires a separate approval reference"
            )
    return RouteDecision(
        capability,
        Transport.API,
        reason,
        policy.api_operation,
        api_authorization,
        provider_mutation_approval,
    )


def record_api_result(
    decision: RouteDecision, *, actual_credits: int, quota_remaining: int
) -> ApiRunEvidence:
    """Validate response-reported cost without hard-coding operation pricing."""

    if decision.transport is not Transport.API or decision.authorization is None:
        raise Portfolio123RoutingError("API evidence requires an API decision")
    if actual_credits < 0 or quota_remaining < 0:
        raise Portfolio123RoutingError("response credit values cannot be negative")
    if actual_credits > decision.authorization.authorized_credit_budget:
        raise Portfolio123RoutingError("actual API cost exceeded the authorized budget")
    return ApiRunEvidence(decision, actual_credits, quota_remaining)
