from __future__ import annotations

import pytest

from quant_engine.portfolio123.capture import (
    CapturePlanError,
    account_capture_targets,
    managed_strategy_capture_targets,
    shared_strategy_capture_targets,
)


def test_shared_capture_plan_is_complete_and_deterministic() -> None:
    first = shared_strategy_capture_targets(123456)
    second = shared_strategy_capture_targets(123456)
    assert first == second
    assert {target.capability for target in first} == {
        "strategy.summary",
        "strategy.trading-system",
        "strategy.transactions",
        "strategy.holdings",
        "strategy.performance",
        "strategy.period-performance",
        "strategy.period-performance-distribution",
    }
    assert all(target.url.startswith("https://www.portfolio123.com/") for target in first)


def test_spa_plans_have_no_hidden_or_default_identifier() -> None:
    assert len(managed_strategy_capture_targets(123456)) == 5
    assert len(account_capture_targets(123456)) == 5
    for invalid in (0, -1, True, 1.5, "123456"):
        with pytest.raises(CapturePlanError):
            shared_strategy_capture_targets(invalid)  # type: ignore[arg-type]
