from __future__ import annotations

from datetime import date
from decimal import Decimal
import json
from pathlib import Path


REPOSITORY_ROOT = Path(__file__).resolve().parents[2]
POLICY_PATH = (
    REPOSITORY_ROOT / "docs" / "decisions" / "0002-portfolio-performance-policy.md"
)
VECTORS_PATH = REPOSITORY_ROOT / "tests" / "fixtures" / "performance_policy_v1.json"


def _vectors() -> dict[str, object]:
    parsed = json.loads(VECTORS_PATH.read_text(encoding="utf-8"))
    assert isinstance(parsed, dict)
    return parsed


def test_policy_is_accepted_read_only_and_contains_fail_closed_rules() -> None:
    policy = POLICY_PATH.read_text(encoding="utf-8")
    required_markers = {
        "**Status:** Accepted for read-only baseline reporting",
        "**Policy version:** `PORT-PERF-1`",
        "`INFERRED_TIMING`",
        "`UNAVAILABLE`",
        "`UNMAPPED`",
        "`DO_NOT_EXPAND_SYSTEMATIC_RISK`",
        "`ELIGIBLE_FOR_REVIEW`",
        "The hard maximum drawdown limit is `15%` of systematic capital only.",
        "Direct-options risk machinery",
        "does not authorize capital movement or trading",
    }
    for marker in required_markers:
        assert marker in policy


def test_policy_vectors_have_one_version_and_separate_evidence_layers() -> None:
    vectors = _vectors()
    assert vectors["schema_version"] == 1
    assert vectors["policy_version"] == "PORT-PERF-1"
    assert vectors["classification"] == "SYNTHETIC"
    assert vectors["evidence_layers"] == [
        "ACCOUNT_LIVE",
        "PROVIDER_LIVE",
        "OFFICIAL_VEHICLE",
        "PREDECESSOR",
        "SHADOW_OOS",
        "BACKTEST",
        "CONCEPTUAL_STRESS",
    ]


def test_hand_calculated_twr_vectors() -> None:
    cases = {item["case_id"]: item for item in _vectors()["twr_cases"]}  # type: ignore[index]
    exact = cases["exact_timestamp_segments"]
    linked = Decimal("1")
    for period in exact["subperiods"]:
        beginning = Decimal(period["beginning_value"])
        ending = Decimal(period["ending_value"])
        linked *= ending / beginning
    assert linked - Decimal("1") == Decimal(exact["expected_return"])
    assert exact["flags"] == []

    date_only = cases["date_only_end_of_day"]
    result = (
        Decimal(date_only["ending_value"]) - Decimal(date_only["net_external_flow"])
    ) / Decimal(date_only["beginning_value"]) - Decimal("1")
    assert result == Decimal(date_only["expected_return"])
    assert date_only["flags"] == ["INFERRED_TIMING"]


def test_hand_calculated_xirr_vector_uses_actual_365_and_investor_signs() -> None:
    case = _vectors()["mwr_cases"][0]  # type: ignore[index]
    first, terminal = case["cash_flows"]
    elapsed = (
        date.fromisoformat(terminal["date"]) - date.fromisoformat(first["date"])
    ).days
    assert case["day_count"] == "ACTUAL_365"
    assert elapsed == 365
    initial = -Decimal(first["investor_cash_flow"])
    ending = Decimal(terminal["investor_cash_flow"])
    xirr = (ending / initial) ** (Decimal("365") / Decimal(elapsed)) - Decimal("1")
    assert xirr == Decimal(case["expected_return"])


def test_hand_calculated_composite_twr_is_beginning_capital_weighted() -> None:
    case = _vectors()["composite_twr_cases"][0]  # type: ignore[index]
    members = case["members"]
    capital = sum(
        (Decimal(item["beginning_capital"]) for item in members), Decimal("0")
    )
    result = (
        sum(
            (
                Decimal(item["beginning_capital"]) * Decimal(item["return"])
                for item in members
            ),
            Decimal("0"),
        )
        / capital
    )
    assert result == Decimal(case["expected_return"])


def test_hwm_and_correlation_one_vectors_fail_closed() -> None:
    cases = {item["case_id"]: item for item in _vectors()["risk_cases"]}  # type: ignore[index]
    hwm_case = cases["complete_hwm_index"]
    assert hwm_case["full_systematic_coverage"] is True
    index = Decimal("1")
    hwm = index
    for daily_return in hwm_case["daily_returns"]:
        index *= Decimal("1") + Decimal(daily_return)
        hwm = max(hwm, index)
    drawdown = Decimal("1") - index / hwm
    assert drawdown == Decimal(hwm_case["expected_drawdown"])

    stress_case = cases["correlation_one_below_limit"]
    stress = sum(
        (
            Decimal(item["weight"]) * Decimal(item["stressed_loss"])
            for item in stress_case["strategies"]
        ),
        Decimal("0"),
    )
    assert stress == Decimal(stress_case["expected_stress"])
    assert stress <= Decimal("0.15")
    assert stress_case["expected_readiness"] == "ELIGIBLE_FOR_REVIEW"

    missing = cases["missing_full_hwm_fails_closed"]
    assert missing["full_systematic_coverage"] is False
    assert missing["expected_hwm"] == "UNAVAILABLE"
    assert missing["expected_readiness"] == "DO_NOT_EXPAND_SYSTEMATIC_RISK"
